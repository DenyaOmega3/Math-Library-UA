\subsubsection{Вкладення Сегре}
Нехай $\psi \colon \mathbb{P}^r \times \mathbb{P}^s \to \mathbb{P}^{(r+1)(s+1)-1}$ задається формулою:
\begin{align*}
[X_0: \ldots :X_r] \times [Y_0: \ldots : Y_s] \mapsto [X_0Y_0: X_0Y_1: \dots X_0Y_s: X_1Y_0: X_1Y_1: \dots X_rY_s].
\end{align*}
\begin{enumerate}[wide=0pt,label={\Roman*.}]
\item $\psi$ \textit{коректно визначене}.\\
Дійсно, масштабуючи точку $[X_0:\ldots:X_r]$ на $\lambda$ та точку $[Y_0:\ldots:Y_s]$ на $\mu$ (\textit{це ми взяли еквівалентні точки}), то звідси $[X_0Y_0: \ldots: X_r Y_s]$ масштабується на $\lambda \mu$, еквівалентність виконується.

\item \textit{$\psi$ ін'єктивне}.\\
Нехай $[X_0Y_0:X_0Y_1:\ldots:X_rY_s] = [A_0B_0:A_0B_1:\ldots:A_rB_s]$. Хочемо довести, що насправді $[X_0:\ldots:X_r] \times [Y_0:\ldots:Y_s] = [A_0:\ldots:A_r] \times [B_0:\ldots:B_s]$. Однак наприклад, однорідні координати
\[
X_0 = X_0 Y_0 = A_0 B_0 = A_0, \qquad Y_0 = X_0 Y_0 = A_0 B_0 = B_0.
\]
Із іншими координатами аналогічно.

\item \textit{$\psi(\mathbb{P}^r \times \mathbb{P}^s)$ задає проєктивний многовид в $\mathbb{P}^{(r+1)(s+1)-1}$}.\\
Для початку розглянемо гомоморфізм $k[Z_{00}: Z_{01}: \ldots : Z_{rs}] \to k[x_0,\dots,x_r,y_0,\dots,y_s]$, що задається як $Z_{ij} \mapsto x_i y_j$. Доведемо, що $\psi(\mathbb{P}^r \times \mathbb{P}^s) = Z(\ker f)$. Можна вже зазначити, що многочлени $u_{ijkl}(Z_{00},\dots,Z_{rs}) = Z_{ij} Z_{kl} - Z_{kj} Z_{il}$ уже потрапляють в $\ker f$.\\
Нехай $P \in Z(\ker f)$, де $P = [Z_{00}: \ldots: Z_{rs}]$. Зокрема розглянемо многочлен $u_{ijkl} \in \ker f$, звідси (TODO:?)
\end{enumerate}

\subsubsection{Проєктивне замикання афінного многовида}
\begin{definition}
Нехай $Y \subset \mathbb{A}^n$ --- афінний многовид. Ототожнимо $\mathbb{A}^n$ із відкритою множиною $U_0 \subset \mathbb{P}^n$ гомоморфізмом $\varphi_0$. Тоді ототожнимо $Y$ та візьмемо його замикання, позначу так само $\Cl Y$. Ось це й називається \textbf{проєктивним замиканням $Y$}.
\end{definition}

\begin{proposition}
Нехай $Y \subset \mathbb{A}^n$ --- афінний многовид та розглянемо його проєктивне замикання $\Cl Y$. Тоді
\begin{align*}
I( \Cl Y) = \left< \beta(I(Y)) \right>,
\end{align*}
де $\beta$ є перетворювачем в однорідний многочлен.
\end{proposition}

\begin{proof}
Для бажаного доведемо, що $Z(\beta(I(Y))) = \Cl Y$.
\begin{enumerate}[wide=0pt,label={\Roman*.}]
\item \textit{$Z(\beta(I(Y)))$ містить $Y$}.\\
Дійсно, нехай $P = [a_0:a_1:\ldots:a_n] \in Y$, цю точку ототожнимо з $\left( \dfrac{a_1}{a_0}, \dots, \dfrac{a_n}{a_0}\right)$. Нехай тепер $f \in \beta(I(Y))$, звідси $f(x_0,\dots,x_n) = x_0^{\deg g} g\left( \dfrac{x_1}{x_0},\dots,\dfrac{x_n}{x_0}\right)$ для деякого $g \in I(Y)$. Проте тоді $g(P) = 0$, внаслідок чого $f(P) = 0$.

\item \textit{$Z(\beta(I(Y)))$ найменша можлива множина}.\\
Нехай $W$ інша замкнена множина, що містить $Y$, тобто $W = Z(\mathfrak{a})$ для деякого однорідного ідеалу $\mathfrak{a}$. Звідси одержимо $I(Y) \supset I(\mathfrak{a})$. Хочемо довести, що $W \supset Z(\beta(I(Y)))$.\\
Нехай $Q \in Z(\beta(I(Y)))$, тобто $f(Q) = 0$ для кожного $f \in \beta(I(Y))$, тобто $x_0^{\deg g} g\left( \dfrac{x_1}{x_0},\dots, \dfrac{x_n}{x_0}\right) = 0$. Якщо ототожнити $Q$ з точокю $P = \left( \dfrac{x_1}{x_0},\dots, \dfrac{x_n}{x_0}\right)$, то одержимо $P \in Z(I(Y)) = Y \subset W$. \qedhere
\end{enumerate}
\end{proof}

\begin{remark}
Слід зауважити, що якщо $f_1,\dots,f_r$ породжують $I(Y)$, то зовсім не обов'язково $\beta(f_1),\dots,\beta(f_r)$ породжують $I(\Cl Y)$.
\end{remark}

\begin{example}
Зокрема розглянемо $Y \subset \mathbb{A}^3$, що задається кривою $(x,y,z) = (t,t^2,t^3)$. Відомо, що $f_1 = y-x^2$ та $f_2 = z-x^3$ породжують $I(Y)$. Однак $\beta(f_1), \beta(f_2)$ не породжують $I(\Cl Y)$, для цього треба розглянути $f_3 = z - xy$ та довести, що $\beta(f_3)$ не розписується як комбінація попередніх двох.
\end{example}

\subsection{Раціональні відображення}
\subsubsection{Основні означення}
\begin{definition}
Нехай $X,Y$ --- алгебраїчні многовиди та $\varphi \colon X \to Y$ --- відображення (\textit{не обов'язково морфізм}). Визначимо пару $\left<U,\varphi|_U\right>$, де $U \subset X$ --- відкрита непорожня множина та $\varphi|_U$ --- морфізм. На таких наборах пар визначимо відношення еквівалентності:
\begin{align*}
\left<U, \varphi|_U\right> \sim \left<V, \varphi|_V\right> \iff \varphi|_U \equiv \varphi|_V \text{ на } U \cap V.
\end{align*}
Тоді клас еквівалентності пар $\left<U,\varphi|_U\right>$ називається \textbf{раціональним відображенням $\varphi \colon X \to Y$}. 
\end{definition}

Для класу еквівалентності (\textit{точніше кажучи, для транзитивності}) потрібна ось ця лема.

\begin{lemma}
Нехай $X,Y$ --- алгебраїчні многовиди та $\varphi, \psi \colon X \to Y$ --- морфізми. Припустимо, існує відкрита непорожня підмножина $U \subset X$, для якої $\varphi|_U = \psi|_U$. Тоді $\varphi \equiv \psi$.
\end{lemma}

\begin{proof}
Розглянемо добуток $\mathbb{P}^n \times \mathbb{P}^n$, який має структуру проєктивного многовида в $\mathbb{P}^{(n+1)(n+1)-1}$ згідно з вкладенням Сегре. Визначимо відображення $\varphi \times \psi \colon X \to \mathbb{P}^n \times \mathbb{P}^n$, яке буде морфізмом (\textit{згідно з теорії категорії}) (TODO: додати про це). Розглянемо діагональ $\Delta = \{ P \times P \mid P \in \mathbb{P}^n \}$, причому $\Delta \subset \mathbb{P}^n \times \mathbb{P}^n$ буде замкненою, оскільки після вкладення Сегре одержимо $\Delta = Z(\{x_i y_j - x_j y_i\}$. Крім того, за припущенням леми $\varphi \times \psi (U) \subset \Delta$. Проте $U$ --- щільна та $\Delta$ --- замкнена, тож якщо взяти замикання з двох сторін, одержимо $\varphi \times \psi(X) \subset \Delta$ --- отримали бажане.
\end{proof}

\begin{definition}
Раціональне відображення $\varphi \colon X \to Y$ називають \textbf{домінуючим}, якщо
\begin{center}
існує $\left<U,\varphi|_U\right>$ таким чином, що $\varphi|_U(U)$ буде щільною в $Y$.
\end{center}
\end{definition}

\begin{remark}
Насправді, якщо існує $\left<U,\varphi|_U\right>$, для якого $\varphi|_U(U)$ буде щільною в $Y$, то тоді й для будь-якого іншого представника $\left<V, \varphi|_V \right>$ щільною в $Y$ буде також $\varphi|_V(V)$.
\bigskip \\
Дійсно, для початку $U,V$ --- відкриті щільні, тому $U \cap V$ --- щільна. Крім того, $\varphi|_V (U \cap V) = \varphi|_U (U \cap V)$ щільна в $\varphi|_U(U)$. Оскільки $\varphi|_U(U)$ щільна в $Y$, то звідси $\varphi|_V(U \cap V)$ --- щільна в $Y$. Нарешті, $\varphi|_V(V) \supset \varphi|_V(U \cap V)$, а взявши замикання з двох сторін, отримаємо $\Cl \varphi|_V(V) \supset Y$, тобто $\varphi|_V(V)$ --- щільна також в $Y$.
\end{remark}

Створимо категорію \vardom, об'єктами яких будуть алгебричні многовиди, а відображеннями між ними будуть домінуючі раціональні відображення (\textit{вправа: довести, що композиція працює}).

\begin{definition}
Ізоморфізмом в категорії \vardom називають \textbf{біраціональним відображенням}. Алгебричні многовиди $X,Y$ називаються \textbf{біраціонально еквівалентними}, якщо існує між ними біраціональне відображення $\varphi \colon X \to Y$. 
\end{definition}

\begin{theorem}
Справедлива ось така еквівалентність категорії, де стрілки змінюють напрямок:
\begin{center}
\vardom $\equiv$ \fgFieldExt .
\end{center}
Більш розгорнуто, для будь-яких двох алгебричних многовидів $X,Y$ існує бієкція
\begin{center}
$\left\{ \substack{\text{домінуючі раціональні} \\ \text{відображення $X \to Y$}} \right\} \leftrightarrow \left\{  \substack{\text{гомоморфізм $K(Y) \to K(X)$} \\ \text{$k$-алгебр}} \right\}$
\end{center}
\end{theorem}

Перш ніж довести теорему, нам будуть необхідні наступні леми.

\begin{lemma}
Нехай $Y \subset \mathbb{A}^n$ --- гіперповерхня, задана многочленом $f(x_1,\dots,x_n) = 0$. Тоді
\begin{align*}
\mathbb{A}^n \setminus Y \cong H,
\end{align*}
де $H \subset \mathbb{A}^{n+1}$ --- гіперповерхня, задана рівнянням $x_{n+1} f(x_1,\dots,x_n) = 1$. Внаслідок чого $\mathbb{A}^n \setminus H$ буде афінним многовидом, причому
\begin{align*}
A(\mathbb{A}^n \setminus Y) = A_f.
\end{align*}
\end{lemma}

\begin{proof}
Визначимо функцію $\varphi \colon H \to \mathbb{A}^n$ ось так:
\begin{align*}
\varphi(a_1,\dots,a_n,a_{n+1}) = (a_1,\dots,a_n).
\end{align*}
Воно задає морфізм, якому ставиться в відповідність гомоморфізм $A(\mathbb{A}^n) \to \mathcal{O}(H)$, тобто $A \to A_f$ (TODO: чому $A_f$?). Побудуємо також $\varphi^{-1} \colon \mathbb{A}^n \setminus Y \to H$ ось так:
\begin{align*}
\varphi^{-1}(a_1,\dots,a_n) = \left( a_1,\dots,a_n, \dfrac{1}{f(a_1,\dots,a_n)} \right).
\end{align*}
Покординатним чином можна довести, що $\varphi^{-1}$ також морфізм. Тоді $\varphi \colon H \to \mathbb{A}^n \setminus Y$ задає ізоморфізм. Внаслідок чого знову за відповідністю $A(\mathbb{A}^n \setminus Y) = A_f$. 
\end{proof}

\begin{proposition}
Будь-який алгебричний многовид має базу
\begin{align*}
\mathcal{B} = \{ \text{відкриті афінні підмножини} \}.
\end{align*}
\end{proposition}

\begin{proof}
Нехай $Y$ --- алгебричний многовид. Хочемо довести, що для будь-якого $P \in Y$ та будь-якого відкритого $U \ni P$ існує відкритий афінний окіл $V$ точки $P$, щоб $V \subset U$.\\
Оскільки $U$ є також алгебричний многовидом, то припустимо, що $U = Y$.\\
Будь-який алгебричний многовид покритий квазі-афінними многовидами, тому припустимо, що $Y$ --- квазі-афінний в $\mathbb{A}^n$.\\
Нехай $Z = \Cl Y \setminus Y$, який замкнений в $\mathbb{A}^n$. Нехай $\mathcal{a} \subset A$ --- ідеал для $Z$. Оскільки $Z$ замкнена та $P \notin Z$, то знайдемо $f \in \mathcal{a}$, для якого $f(P) \neq 0$. Нехай тоді $H$ --- гіперповерхня, що задається $f = 0$ на $\mathbb{A}^n$. Звідси $Z \subset H$, проте $P \notin H$. Значить, $P \in Y \setminus (Y \cap H)$, остання множина відкрита в $Y$.\\
Також $Y \setminus (Y \cap H)$ є замкненою в $\mathbb{A}^n \setminus H$, що є афінним за попередньою лемою. Покладемо $V = \mathbb{A}^n \setminus H \ni P$ та одержимо $V \subset \mathbb{A}^n = U$. 
\end{proof}
