\section{Границі числової послідовності}
\subsection{Означення границі}
\begin{problem}
За означенням довести наступні границі:
\begin{enumerate}[nosep,wide=0pt,label={\arabic*)}]
\item $\displaystyle\lim_{n \to \infty} \dfrac{(-1)^n}{n} = 0$;
\item $\displaystyle\lim_{n \to \infty} \dfrac{-4n+2025}{2020+2n} = -2$;
\item $\displaystyle\lim_{n \to \infty} \dfrac{n^2+4}{2n^2-5} \sin \dfrac{\pi n}{4} = \dfrac{1}{2}$.
\end{enumerate}
\end{problem}

\begin{problem}
За означенням довести наступні границі (вказівка: користуючись різними нерівностями):
\begin{enumerate}[nosep,wide=0pt,label={\arabic*)}]
\item $\displaystyle\lim_{n \to \infty} \sqrt[n]{n} = 1$;
\item $\displaystyle\lim_{n \to \infty} n q^n = 0$ (за умовою, що $|q| < 1$);
\item $\displaystyle\lim_{n \to \infty} \dfrac{1}{2} \cdot \dfrac{3}{4} \cdots \dfrac{2n-1}{2n} = 0$ (вказівка: $\dfrac{1}{2} \cdot \dfrac{3}{4} \cdots \dfrac{2n-1}{2n} < \dfrac{1}{\sqrt{2n+1}}$);
\end{enumerate}
\end{problem}

\subsection{Обчислення границі}
\begin{problem}
Обчислити границі числових послідовностей:
\begin{enumerate}[nosep,wide=0pt,label={\arabic*)}]
\item $\displaystyle\lim_{n \to \infty} \left( \dfrac{1+2+\dots+n}{n+1} - \dfrac{n}{2} \right)$;
\item $\displaystyle\lim_{n \to \infty} \dfrac{2^{n+2} + 3^{n+3}}{2^n + 3^n}$;
\item $\displaystyle\lim_{n \to \infty} \dfrac{1+a+\dots+a^n}{1+b+\dots+b^n}$ (за умовою, що $|a| < 1,\ |b|<1$);
\item $\displaystyle\lim_{n \to \infty} (\sqrt{n+1} - \sqrt{n})$;
\item $\displaystyle\lim_{n \to \infty} \dfrac{\sin n!}{n^{2025}}$;
\end{enumerate}
\end{problem}

\subsection{Критерій Коші}
\begin{example}
Довести збіжність послідовності $\{x_n, n \geq 1\}$, що задається такою формулою: $x_n = \dfrac{\cos 1!}{1 \cdot 2} + \dfrac{\cos 2!}{2 \cdot 3} + \dots + \dfrac{\cos n!}{n(n+1)}$.
\end{example}

\begin{example}
Довести, що послідовність $x_n = 1 + \dfrac{1}{2} + \dfrac{1}{3} + \dots + \dfrac{1}{n}$ розбігається.\\
\textit{Цей приклад дуже важливий для теорії числових рядів.}
\end{example}

\begin{example}
Послідовність $\{x_n,n \geq 1\}$ має \textbf{обмежену варіацію}, якщо існує таке $C > 0$, для якого
\begin{align*}
\forall n \geq 2: |x_2 - x_1| + |x_3 - x_2| + \dots + |x_n - x_{n-1}| < C.
\end{align*}
Довести, що будь-яка послідовність обмеженої варіації збіжна.
\end{example}
