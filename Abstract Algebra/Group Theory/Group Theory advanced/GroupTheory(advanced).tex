\subsection{Лема Гурса}
\begin{definition}
Нехай $G_1,G_2$ --- групи, розглянемо підгрупу $H \leq G_1 \times G_2$.\\
підгрупа $H$ називається \textbf{підпрямим добутком} $G_1,G_2$, якщо
\begin{center}
$p_1 \colon H \to G_1$ та $p_2 \colon H \to G_2$ --- сюр'єктивні проєкції.
\end{center}
\end{definition}

\begin{theorem}
Нехай $G,G'$ --- групи та $H \leq G \times G'$ --- підпрямий добуток. Позначимо $N = \ker p_2$ та $N' = \ker p_1$. Тоді образом $H$ в групі $G/_N \times G'/_{N'}$ буде графіком ізоморфізма $G/_N \cong G'/_{N'}$.
\end{theorem}

\begin{proof}
\begin{enumerate}[wide = 0pt, topsep=0pt,label={\Roman*.}]
\item \textit{$N \triangleleft G$ та $N' \triangleleft G'$}.\\
Оскільки $p_2$ --- гомоморфізм, то $N \triangleleft H$. Крім того, для кожного $g \in G$ існує $h = (g,g') \in H$ так, що $p_1(h) = g$. Таким чином, $p_1(N) \triangleleft G$, оскільки
\[
g p_1(N) = p_1(h) p_1(N) = p_1(hN) = p_1(Nh) = p_1(N)g.
\]
Отже, $N \triangleleft G \times \{e'\}$, просто тому що
\[
(g,e')N = (g,e') (p_1(N) \times \{e'\}) = gp_1(N) \times \{e'\} = p_1(N)g \times \{e'\} = (p_1(N) \times \{e'\})(g,e') = N(g,e').
\]
Знаючи, що $G \cong G \times \{e'\}$, можемо сприймати $N$ як нормальну підгрупу до $G$.\\
Аналогічно одержимо, що $N' \triangleleft \{e\} \times G'$.

\item \textit{Образом $H$ в $G/_N \times G'/_{N'}$ буде графіком}.\\
Нехай відображення $H \to G/_N \times G'/_{N'}$ визначений як
\[
(g,g') \to (gN, g'N').
\]
Образом $H$ тут буде множина $\{ (gN, g'N') \mid (g,g') \in H \}$. Оскільки $H \to G/_N$ сюр'єктивне, то можна визначити відображення $G/_N \to {G'}/_{N'}$ як $gN \mapsto g'N'$, внаслідок чого $\{ (gN, g'N') \mid (g,g') \in H \}$ буде графіком.

\end{enumerate}
\end{proof}

\section{Підгрупи Хола}
\subsection{Характеристична підгрупа}
\begin{definition}
Нехай $G$ --- група. Тоді підгрупа $H \leq G$ називається \textbf{характеристичною в $G$}, якщо для будь-якого автоморфізма $\varphi \colon G \to G$ справедлива рівність
\[
\varphi(H) = H.
\]
Позначення: $H \charac G$.
\end{definition}

\begin{proposition}
Нехай $\varphi(H) \leq H$ для будь-якого автоморфізму $\varphi \colon G \to G$. Тоді $H \charac G$.
\end{proposition}

\begin{proof}
Дійсно, $\varphi,\varphi^{-1}$ будуть автоморфізмами групи $G$. Тоді за умовою $\varphi(H) \leq H$ та $\varphi^{-1}(H) \leq H \implies H = \varphi \varphi^{-1}(H) \leq \varphi(H)$, тому звідси $\varphi(H) = H$.
\end{proof}

\begin{proposition}
Нехай $H \charac G$. Тоді $H \triangleleft G$.
\end{proposition}

\begin{proof}
Справді, спряження $x \mapsto axa^{-1}$ буде автоморфізмом групи $G$ для кожного $a \in G$. Зокрема оскільки $H \charac G$, то звідси для кожного $x \in H$ матимемо $axa^{-1} \in H$.
\end{proof}

\begin{remark}
Зворотне твердження не є вірним, якщо не накласти додаткових умов.
\end{remark}

\begin{example}
Розглянемо групу $V \cong \mathbb{Z}_2 \times \mathbb{Z}_2$ та його підгрупу $\mathbb{Z}_2 \triangleleft V$. Можна підібрати автоморфізм $\varphi \colon V \to V$ як $(x,y) \mapsto (y, x+y)$, для якого $\varphi(\mathbb{Z}_2 \times \{0\}) \neq \mathbb{Z}_2 \times \{0\}$.
\end{example}

\begin{proposition}
\label{hall_normal_subgroup_is_characteristic}
Нехай $G$ --- скінченна група, $H \triangleleft G$, причому $\gcd(|H|, [G:H]) = 1$. Тоді $H \charac G$.
\end{proposition}

\begin{proof}
Зафіксуємо автоморфізм $\varphi \colon G \to G$. Хочемо (\textit{і цього досить}) довести, що $\varphi(H) \subset H$.\\
Нехай $y \in \varphi(H)$, тобто $y = \varphi(x)$ при деякому $x \in H$. Щоб довести, що $y \in H$, покажемо рівність $yH = H$, або просто доведемо, що $\ord(yH) = 1$.\\
Зауважимо, що $\ord(yH) \mid |G/_H| = [G:H]$. Проте також $\ord(yH) \mid \ord g = \ord \varphi(x) = \ord x \mid |H|$. Тобто $\ord(yH)$ буде спільним дільником чисел $|H|,\ [G:H]$, звідси $\ord(yH) = 1$.
\end{proof}

\begin{proposition}
Нехай $H \charac K$ та $K \charac G$. Тоді $H \charac G$ (\textit{на відміну від нормальних підгруп})
\end{proposition}

\begin{proof}
Дійсно, нехай $\varphi$ --- автоморфізм $G$. Оскільки $K \charac G$, то $\varphi(K) = K$, тому звуження $\varphi|_K$ буде автоморфізмом $K$. Оскільки $H \charac K$, то звідси $\varphi(H) = (\varphi|_K)(H) = H$.
\end{proof}

\begin{proposition}
Нехай $H \charac K$ та $K \triangleleft G$. Тоді $H \triangleleft G$.
\end{proposition}

\begin{proof}
Дійсно, нехай $a \in G$ та $\varphi$ --- автоморфізм, що описує спряження по $a$. Оскільки $K \triangleleft G$, то $\varphi|_K$ буде автоморфізмом $K$. Оскільки $H \charac K$, то одержимо $(\varphi|_K)(H) \leq H$. Тобто якщо $h \in H$, то звідси $aha^{-1} = \varphi(h) \in H$.
\end{proof}

Тепер розглянемо кілька прикладів характеристик, які можуть виникати, їх я залишу в вигляді тверджень.

\begin{proposition}
$Z(G) \charac G$ (\textit{миттєво доводиться}).
\end{proposition}

\begin{proposition}
Нехай $G$ --- циклічна група. Тоді для будь-якої підгрупи $H$ маємо $H \charac G$.
\end{proposition}

\begin{proof}
Дійсно, нехай $\varphi \colon G \to G$ --- автоморфізм та $H \leq G$ (\textit{яка буде також циклічною}). Нам треба довести, що $\varphi(H) \leq H$, тобто якщо $H = \left<a\right>$, то хочемо, щоб $\left<\varphi(a)\right> \subset \left<a\right>$. Але якщо $\varphi(a) = b$ для деякого $b \in H$, то звідси $\varphi(a) = a^k$ при деякому $k \in \mathbb{Z}$.
\end{proof}

\begin{proposition}
Нехай $G$ --- група. Тоді $G^{(i)} \charac G$.\\
(\textit{це ще один спосіб довести, що похідні вищих порядків є нормальними підгрупами})
\end{proposition}

\begin{proof}
Доведення буде за МІ по $i$.
\begin{enumerate}[wide=0pt,nosep,label={\Roman*.}]
\item \textit{База індукції}. При $i = 1$ маємо автоморфізм $\varphi \colon G \to G$, тоді відомо, що $\varphi(G') = G'$, оскільки $\varphi$ --- сюр'єктивний. Отже, $G' \charac G$.
\item \textit{Припущення індукції}. $G^{(i-i)} \charac G$ для деякого $i$.
\item \textit{Крок індукції}. Розглянемо $\varphi \colon G^{(i-1)} \to G^{(i-1)}$, але аналогічно тоді $\varphi(G^{(i)}) = \varphi((G^{(i-1)})') = (G^{(i-1)})' = G^{(i)}$, тобто ми маємо $G^{(i)} \charac G$. За припущенням МІ $G^{(i-i)} \charac G$, внаслідок чого $G^{(i)} \charac G$.
\end{enumerate}
МІ доведено.
\end{proof}

\subsection{Мінімальна нормальна підгрупа, характеристично прості групи}
\begin{definition}
Нехай $G$. Підгрупа $H$ називається \textbf{мінімальною нормальною підгрупою} для $G$, якщо
\begin{center}
$H \neq \{e\}$, а також не існує $K \triangleleft G$ так, щоб $\{e\} < K < H$.
\end{center}
Позначення: $H \triangleleft_{\text{min}} G$.
\end{definition}

\begin{remark}
Якщо $G$ --- нетривіальна скінченна група, то мінімальна нормальна підгрупа завжди існує. Для нескінченних груп це не завжди правильно (\textit{вказівка: $\mathbb{Z}$}).
\end{remark}

\begin{theorem}
Нехай $G$ --- скінченна розв'язна група. Тоді будь-яка мінімальна нормальна підгрупа --- елементарна абелева.
\end{theorem}

\begin{proof}
Оскільки $G$ --- скінченна, оберемо підгрупу $V \triangleleft_{\text{min}} G$. Відомо, що $V' \charac V$, тоді одержимо, що $V' \triangleleft G$. У силу мінімальності підгрупи маємо або $V' = \{e\}$, або $V' = V$. Проте останнє неможливе, оскільки $V$ буде також розв'язною підгрупою. Значить, $V' = \{e\}$, тобто $V$ --- абелева.\\
Оберемо нетривіальну $p$-підгрупу Сілова $P \triangleleft V$ (\textit{нормальність в силу абелевості}). Тоді оскільки $\gcd(|P|, [V:P]) = 1$, то одержимо $P \charac V$, але звідси тоді $P \triangleleft G$. Знову або $P = \{e\}$, або $P = V$. Для $p$-підгруп Сілова перше неможливе. Значить, $V$ буде $p$-групою.\\
Нарешті, зауважимо, що $\{x \in V: x^p = 1\} \charac V$, тобто знову $\{x \in V: x^p = 1 \} \triangleleft G$, звідси одержимо $V = \{x \in V: x^p = 1 \}$, тобто всі елементи (\textit{окрім $e$}) мають порядок $p$, тобто це елементарна абелева група.
\end{proof}

\begin{definition}
Група $G$ називається \textbf{характеристично простою}, якщо
\begin{center} 
лише $\{e\},G$ є характеристичними підгрупами $G$.
\end{center}
\end{definition}

\begin{theorem}
Нехай $G$ --- скінченна, характеристично проста група. Тоді $G$ або проста, або є прямим добутком ізоморфних простих груп.
\end{theorem}

\begin{proof}
Отже, оберемо підгрупу $H \triangleleft_{\text{min}} G$. Позначимо $H = H_1$ та розглянемо всі підгрупи $G$ вигляду $H_1 \times H_2 \times \dots \times H_n$ при $n \geq 1$, де всі $H_i \triangleleft G$ та $H_i \cong H$. Позначимо $M$ за підгрупу щойно описаного вигляду, яка має найбільший можливий порядок. Доведемо $M = G$, якщо покажемо, що $M \charac G$.\\
Насправді, якщо $\varphi \colon G \to G$ --- автоморфізм групи $G$, то нам буде досить довести, що $\varphi(H_i) \leq M$ для кожного $i$. Також зауважимо, що $\varphi(H_i) \cong H = H_1$.\\
Доведемо, що $\varphi(H_i) \triangleleft G$. Нехай $a \in G$, тобто $a = \varphi(b)$ для деякого $b \in G$. Але тоді
\[
a \varphi(H_i) a^{-1} = \varphi(b) \varphi(H_i) \varphi(b)^{-1} = \varphi(a H_i a^{-1}) \overset{H_i \triangleleft G}{\leq} \varphi(H_i).
\]
!Тепер припустимо, що $\varphi(H_i) \not\leq M$, тоді $\varphi(H_i) \subsetneq M \not\leq \varphi(H_i)$, але звідси
\[
|\varphi(H_i) \cap M| < |\varphi(H_i)| = |H|.
\]
Однак $\varphi(H_i) \cap M \triangleleft G$ (\textit{підгрупа $M \triangleleft G$, адже кожна компонента добутку нормальна}), тому за мінімальністю нормальної підгрупи $\varphi(H_i) \cap M = \{e\}$. Значить, група $M \cdot \varphi(H_i) \cong M \times \varphi(H_i)$ --- це група вигляду, що й $M$, але порядком більша за $M$ --- суперечність!\\
Нарешті, $H$ --- проста, бо якби $N \triangleleft H$, то було би $N \triangleleft M = H_1 \times \dots \times H_n = G$, однак це суперечить мінімальності нормальної підгрупи.
\end{proof}

\subsection{Теореми Хола}
\begin{theorem}[Аргумент Фраттіні]
Нехай $G$ --- скінченна група та $K \triangleleft G$. Припустимо, що $P$ --- підгрупа Сілова $K$. Тоді $G = K \cdot N_G(P)$.
\end{theorem}

\begin{proof}
Нехай $g \in G$, тоді зауважимо, що $gPg^{-1} \leq gKg^{-1} = K$, просто тому що $K \triangleleft G$. Звідси $gPg^{-1}$ також є підгрупою $K$, причому теж підгрупа Сілова, тому вони спряжені, тобто існує $k \in K$, для якого $kPk^{-1} = gPg^{-1}$. Внаслідок чого $P = (k^{-1}g)P(k^{-1}g)^{-1}$, тобто $k^{-1}g \in N_G(P)$. Тому $g = k(k^{-1}g) \in K \cdot N_G(P)$.
\end{proof}

\begin{theorem}[Теорема Хола]
Нехай $G$ --- розв'язна група порядка $ab$ при $\gcd(a,b) = 1$. Тоді $G$ містить підгрупу порядка $a$. Крім того, будь-які дві підгрупи порядка $a$ є спряженими.\\
(\textit{певного роду дана теорема є узагальненням перших двох теорем Сілова})
\end{theorem}

Доведення буде проводитися за МІ по $|G|$, напишемо це збоку.
\begin{enumerate}[nosep,wide=0pt,label={\Roman*.}]
\item \textit{База індукції}. При $|G| = 1$ взагалі нічого цікавого.
\item \textit{Припущення індукції}. Теорема Хола виконується для груп порядка, що менша за $|G|$.
\item \textit{Крок індукції}. Розглянемо групу $G$ порядка $ab$. Передусім зауважимо, що всі $H \leq G$ мають порядок $a'b'$, де $a' \mid a,\ b' \mid b$. І ось далі доведення теореми розіб'ю на дві можливі леми.
\end{enumerate}

\begin{lemma}
Нехай $G$ --- розв'язна група порядка $ab$ при $\gcd(a,b) = 1$. Припустимо, що існує підгрупа $H \triangleleft G$ порядка $a'b'$, причому $b \nmid |H|$ (\textit{іншими словами, $b' < b$}). Тоді $G$ містить підгрупу порядка $a$. Крім того, будь-які дві підгрупи порядка $a$ є спряженими.
\end{lemma}

\begin{proof}
\begin{enumerate}[topsep=0pt,wide=0pt,label={\roman*.}]
\item \textit{Існування}.\\
Розглянемо тоді групу $G/_H$ порядка $\dfrac{a}{a'} \dfrac{b}{b'}$ (\textit{ці дроби тепер взаємно прості}), яка також розв'язна. Оскільки $|G/_H| < |G|$, то за припущенням індукції $G/_H$ містить підгрупу $A/_H$ порядка $\dfrac{a}{a'}$. Однак тоді $|A| = \dfrac{a}{a'} |H| = ab' < ab = |G|$. Оскільки $A$ розв'язна, то вона містить підгрупу порядка $a$ знову за припущенням МІ.

\item \textit{Спряженість цих підгруп}.\\
Нехай $A,A'$ --- спряжені підгрупи $G$ порядка $a$. Обчислимо значення $|AH|$ та $|A'H|$.\\
Оскільки $AH \leq G$, то за теоремою Лагранжа $|AH| = \alpha \beta$ при $\alpha \mid a,\ \beta \mid b$. Проте оскільки $\gcd(a,b) = 1$ та $A \leq AH$, то знову за теоремою Лагранжа $a \mid \alpha \beta$, причому $a = \alpha$. Аналогічно $H \leq AH$, тому $a'b' \mid \alpha \beta$, тоді $b' \mid \beta$. За другою теоремою про ізоморфізм (\textit{або за формулою про добуток підгруп}) $|AH| \mid aa'b'$, тому $\beta \mid b'$. Значить, $|AH| = ab'$.\\
Аналогічними міркуваннями можна обчислити $|A'H| = ab'$.\\
Отже, $AH/_H,\ A'H/_H$ будуть підгрупами $G/_H$ порядка $\dfrac{a}{a'}$. Оскільки $|G/_H| = \dfrac{a}{a'} \dfrac{b}{b'}$, ці підгрупи будуть спряженими за припущенням МІ. Тобто $xH \cdot AH/_H \cdot (xH)^{-1} = A'H/_H$. Але звідси (\textit{як відомо з базової теорії груп}) $x AH x^{-1} = A'H$, внаслідок чого $xAx^{-1},\ A'$ є підгрупами $A'H$ порядка $a$, а тому вони спряжені за припущенням МІ. \qedhere
\end{enumerate}
\end{proof}

\begin{lemma}
Нехай $G$ --- розв'язна група порядка $ab$ при $\gcd(a,b) = 1$. Припустимо, що для будь-якої підгрупи $H \triangleleft G$ порядка $a'b'$ маємо $b \mid |H|$ (\textit{іншими словами, $b' = b$}). Тоді $G$ містить підгрупу порядка $a$. Крім того, будь-які дві підгрупи порядка $a$ є спряженими.
\end{lemma}

\begin{remark}
Серед всіх нормальних підгруп оберемо $H \triangleleft_{\text{min}} G$, але оскільки $G$ --- розв'язна, то $H$ буде елементарною абелевою, тобто є $p$-групою. Значить, $|H| = p^m$ (\textit{але тоді звідси $b = p^m$}). Отримали $|G| = ap^m$, а підгрупа $H$ стане $p$-підгрупою Сілова для $G$. Оскільки $H \triangleleft G$, то така підгрупа Сілова єдина. Отже, насправді кажучи, другий випадок можна було написати так:\\
\textit{''Припустимо $|G| = ap^m$ при $p \nmid a$ та $G$ має нормальну абелеву єдину $p$-підгрупу Сілова $H \triangleleft_{\text{min}} G$.''}
\end{remark}

\begin{proof}
\begin{enumerate}[topsep=0pt,wide=0pt,label={\roman*.}]
\item \textit{Існування}.\\
Розглянемо ще раз групу $G/_H$ порядка $a$, яка все ще розв'язна. Оберемо $K/_H \triangleleft_{\text{min}} G/_H$, звідси $|K/_H| = q^n$ для деякого простого $q \neq p$ (\textit{аналогічними міркуваннями, що в зауваженні вище}), тому $|K| = p^m q^n$. Нехай $Q$ --- $q$-підгрупа Сілова для $K$. Оскільки $H \cap Q = \{e\}$, то звідси $K = HQ$. Позначимо $N^* = N_G(Q)$ та $N = N^* \cap K = N_K(Q)$. Хочу довести, що $|N^*| = a$.\\
За аргументом Фраттіні $G = KN^*$. Проте зауважимо, що $|N^*| = \dfrac{|G| \cdot |N|}{|K|}$, просто тому що за ІІ теоремою про ізоморфізм
\[
G/_K = KN^*/_K \cong N^*/_{N^* \cap K} = N^*/_N.
\]
Однак також $K = HQ$ та $Q \leq N \leq K$ дає нам $K = HN$. Звідси $|K| = |HN| = \dfrac{|H| \cdot |N|}{|H \cap N|}$, а тому
\[
|N^*| = \dfrac{|G| \cdot |N|}{|K|} = \dfrac{|G| \cdot |N| \cdot |H \cap N|}{|H| \cdot |N|} = \dfrac{|G|}{|H|} \cdot |H \cap N| = a |H \cap N|.
\]
Для того, щоб $|N^*| = a$ за нашим бажанням, треба довести, що $H \cap N$ буде тривіальною групою, для цього доведемо дві штуки.
\begin{itemize}
\item $H \cap N \leq Z(K)$.\\
Нехай $x \in H \cap N$ та розглянемо будь-який $k \in K$, тобто $k = hs$ при $h \in H,\ s \in Q$. Хочемо довести, що $x \in Z(K)$, тобто $x$ комутує з $k$. Оскільки $x$ комутує з $h$ (\textit{бо група $H$ абелева}), то лише треба довести, що $x$ комутує з $s$.\\
Розглянемо комутатор $[x,s]$ та зауважимо, що $[x,s] = (xsx^{-1})s^{-1} \in Q$, оскільки $x \in N = N_K(Q)$; але й також $[x,s] = x(sx^{-1}s^{-1}) \in H$, оскільки група $H$ нормальна. Значить, $[x,s] \in Q \cap H = \{e\}$, адже тут різні підгрупи Сілова. Тобто $[x,s] = e \iff xs = sx$.

\item $Z(K) = \{e\}$.\\
!Припустимо, що $Z(K)$ --- нетривіальна група. Оскільки $Z(K) \charac K$ та $K \triangleleft G$, то звідси $Z(K) \triangleleft G$. Тоді $H \leq Z(K)$, як найменша нормальна підгрупа для $G$. Доведемо нижче, що $Q \charac K$, тоді оскільки $K \triangleleft G$, то одержимо $Q \triangleleft G$. Значить, $H \leq Q$ як найменша нормальна підгрупа для $G$ --- суперечність!\\
Отже, маємо $K = HQ$. Зауважимо, що $H \charac K$, оскільки $\gcd(|H|, [K:H]) = 1$ та $H \triangleleft K$. Тепер розглянемо автоморфізм $\varphi \colon K \to K$, звідси $\varphi(K) = \varphi(HQ) = \varphi(H) \varphi(Q) = H \varphi(Q)$. Крім того, $H \cap \varphi(Q) = \varphi(H) \cap \varphi(Q) = \varphi(H \cap Q) = \varphi(\{e\}) = \{e\}$. Отже, $|\varphi(Q)| = q^n$, тобто $\varphi(Q)$ також буде $q$-підгрупою Сілова для $K$. Звідси $\varphi(Q) = xQx^{-1}$, причому $x = hq$. Оскільки $H \leq Z(K)$, то елемент $h$ комутує з будь-яким елементом з $Q$, внаслідок чого $\varphi(Q) = Q$. Це й доводить, що $Q \charac K$.
\end{itemize}
Разом одержимо, що $|H \cap N| \leq |Z(K)| = 1$, тому одержимо бажане $|N^*| = a$.

\item \textit{Спряженість цих підгруп}.\\
Прпиустимо, що $A$ --- інша підгрупа $G$ порядка $a$. Оскільки $|A| \mid |AK|$ та $|K| \mid |AK|$, тобто $a \mid |AK|$ та $p^m \mid p^m q^m \mid |AK|$, то тоді отримаємо $|AK| = ab = |G|$, внаслідок чого $AK = G$. Отже,
\[
G/_K = AK/_K \cong A/_{A \cap K}
\]
за ІІ теоремою про ізоморфізм, що дає нам $|A\cap K| = q^n$. За ІІ теоремою Сілова $Q$ та $A \cap K$ --- спряжені. Тому $N_G(Q)$ та $N_G(A \cap K)$ --- також спряжені. Утім, насправді кажучи, $N_G(A \cap K) = A$: оскільки $A \cap K \triangleleft A$, то звідси $A \leq N_G(A \cap K)$, причому $|N_G(A \cap K)| = |N^*| = a$. \qedhere
\end{enumerate}
\end{proof}

МІ доведено теорему Хола (\textit{насправді, МІ використовували лише для першої леми}).

\begin{definition}
Нехай $G$ --- скінченна група.\\
Підгрупа $H \leq G$ називається \textbf{підгрупою Хола}, якщо 
\begin{align*}
\gcd(|H|, [G:H]) = 1.
\end{align*}
\end{definition}

\begin{definition}
Нехай $\pi$ --- множина простих чисел. Тоді група $G$ називається \textbf{$\pi$-групою}, якщо
\begin{align*}
\forall g \in G: \ord(g) = \displaystyle\prod_{p \in \pi} p^{\alpha},\ \alpha \geq 0.
\end{align*}
\end{definition}

\begin{remark}
Цілком ясно, що коли $\pi = \{p\}$, то тоді всі $\pi$-групи перетворюються в стандартні $p$-групи.
\end{remark}

\begin{remark}
Теорема Хола каже наступне: якщо $G$ --- скінченна розв'язна група, а $\pi$ --- множина простих чисел, де кожен $\pi \ni p \mid |G|$ то існує $\pi$-підгрупа Хола. Причому дві $\pi$-підгрупи Хола мають бути спряженими.
\end{remark}

\begin{remark}
Будь-яка $p$-підгрупа Сілова буде $\{p\}$-підгрупою Хола.
\end{remark}

\begin{example}
Розглянемо групу $A_5$ та множину $\pi = \{3,5\}$. Оскільки $|A_5| = 60$. Якщо припустити, що існує $\pi$-підгрупа Хола $H$, то тоді $|H| = 3^\alpha 5^\beta$, у неї лише є варіант $|H| = 15$. Значить, $[A_5:H] = 4$, але оскільки $A_5$ --- проста, то за наслідком загальної теореми Келі існує ін'єкція $A_5 \hookrightarrow S_4$, що неможливо.
\end{example}

\begin{remark}
Уже було доведено, що якщо $G$ --- скінченна група та $H \triangleleft G$ --- підгрупа Хола, то $H \charac G$ (\prpref{hall_normal_subgroup_is_characteristic}).
\end{remark}

\begin{definition}
Нехай $G$ --- група порядка $ap^n$ при $p \nmid a$. Тоді підгрупа порядка $a$ називається \textbf{$p$-доповненням групи $G$}.
\end{definition}

\begin{remark}
За теоремою Хола якщо $G$ --- скінченна та розв'язна, то для кожного простого $p$ існує $p$-доповнення.
\end{remark}

\begin{remark}
Якщо $|G| = p^m q^n$ (\textit{а така група розв'язна}), то $p$-доповненням тут буде $q$-підгрупа Сілова.
\end{remark}

\begin{theorem}[Теорема Хола (зворотне)]
Нехай $G$ --- скінченна група, яка містить $p$-доповнення для кожного $p$. Тоді $G$ --- розв'язна.
\end{theorem}

\begin{proof}
Тут будемо доводити за МІ по $|G|$. Скажемо, що для груп порядка меншою за $|G|$ зворотне теореми Хола виконується. Припустимо, що $G$ більше не буде розв'язною. Розіб'ємо на два випадки:
\begin{enumerate}[wide=0pt,label={\Roman*.}]
\item \textit{$G$ --- непроста група}.\\
Отже, оберемо нормальну підгрупу $\{e\} \triangleleft N \triangleleft \{G\}$. Нехай $H$ буде $p'$-підгрупою Хола $G$ (\textit{це типу наше $p$-доповнення до $G$}). Хочетсья довести зараз, що $H \cap N$ буде $p'$-підгрупою Хола для $N$, а $HN/_N$ буде $p'$-підгрупою Хола для $G/_N$. \\
Нехай $|G| = ap^m$, тоді $|H| = a$ як $p$-доповнення. Також одержимо $|N| = a' p^n$, де $a' \mid a$ та $n \leq m$. Зауважимо, що $|H| \mid |HN|$, тобто $a \mid |HN|$. Оскільки $HN \leq G$, то тоді $|HN| = ap^k$ при $k \leq m$. Однак оскільки ще $|N| \mid |HN|$, тобто $a' p^n \mid ap^k$, то тоді $p^n \mid p^k$, тобто $n \leq k \leq m$. Значить, 
\[
|H \cap N| = \dfrac{|H| |N|}{|HN|} = \dfrac{a \cdot a' p^n}{a p^{k}} = a' p^{n-k}.
\]
Залишилося довести, що $n-k = 0$. Однак відомо, що $|H \cap N| \mid |H|$, тобто $a' p^{n-k} \mid a$, тобто $p^{n-k} \mid \dfrac{a}{a'}$, але ось це якраз можливо лише при $n = k$.\\
Отже, $|H \cap N| = a'$, тобто $H \cap N$ буде дійсно $p$-підгрупою Хола для $N$. Крім того, автоматично $HN/_N$ стане $p$-підгрупою Хола для $G/_N$. Оскільки $|N|, |G/_N| < |G|$, то вони розв'язні за припущенням МІ. Проте звідси $G$ стане розв'язною --- суперечить.

\item \textit{$G$ --- проста група}.\\
Нехай $|G| = p_1^{e_1} \dots p_n^{e_n}$. Для кожного $i$ оберемо $p_i'$-підгрупу Хола $H_i$. Тоді $[G:H_i] = p_i^{e_i}$, а також $|H_i| = p_1^{e_1} \underbrace{\dots}_{\text{без $i$}} p_n^{e_n}$. Позначимо $D = H_3 \cap \dots \cap H_n$, звідси випливає $[G:D] = p_3^{e_3} \dots p_n^{e_n}$ та $|D| = p_1^{e_1} p_2^{e_2}$ (TODO: обдумати, це вправа 3.31). За теоремою Бернсайда $D$ --- розв'язна. Оберемо $N \triangleleft_{\text{min}} D$, яка стане елементарною абелевою (\textit{скажімо, $p_1$-підгрупа}). Також маємо $[G:H_2] = p_2^{e_2} p_3^{e_3} \dots p_n^{e_n}$, звідси $|D \cap H_2| = p_1^{e_1}$ та тому $D \cap H_2$ буде $p_1$-підгрупою Сілова для $D$. Значить, $N \triangleleft D \cap H_2$ (TODO: ясно, але обдумати), внаслідок чого $N \leq H_2$. Проте аналогічно $|D \cap H_1| = p_2^{e_2}$, значить $G = H_2(D \cap H_1)$.\\
Тобто якщо $g \in G$, то тоді $g = hd$ при $h \in H_2$ та $d \in D \cap H_1$. Взявши $x \in N$, одержимо, що $gxg^{-1} = hdxd^{-1}h^{-1} = hyh^{-1} \in H_2$, причому $y \in N$, оскільки $N \triangleleft D$. Звідси $N^G \leq H_2 < G$, причому $N \triangleleft G$ --- власна підгрупа --- суперечність. \qedhere
\end{enumerate}
\end{proof}

\begin{remark}
Нехай $\pi = \{3,5\}$. Тоді зауважимо, що $\mathbb{Z}_3,\ \mathbb{Z}_5$ будуть максимальними $\pi$-підгрупами $S_5$, просто тому що $S_5$ не містить підгрупу порядка $15$ (\textit{а це типу наступний більший можливий порядок для двох груп}).\\
Важливо, що ці підгрупи $\mathbb{Z}_3,\mathbb{Z}_5$ неізоморфні, а тому й неспряжені.
\end{remark}

\begin{remark}
Нехай $\pi = \{2,5\}$, тоді максимальна $\pi$-підгрупа групи $S_5$ існує, але вона не є $\pi$-підгрупою Хола. Можливі порядки підгрупи $S_5$ є $\{2,4,5,8,10,20,40\}$. Група порядка $40$ існувати не може, інакше індекс буде $3$.
\end{remark}

\section{Більше фактів про $p$-групи}
\subsection{Деякі леми}
\begin{lemma}
Нехай $G$ --- група та $x,y \in G$ такі, що комутують з $[x,y]$. Тоді
\begin{align*}
[x,y]^n = [x^n,y] = [x,y^n], \qquad n \in \mathbb{Z}; \\
(xy)^n = [y,x]^{C_n^2} x^n y^n, \qquad n \geq 0.
\end{align*}
\end{lemma}

\begin{proof}
\begin{enumerate}[topsep=0pt,wide=0pt,label={\Roman*.}]
\item \textit{Доведення першої формули}.\\
Розглянемо випадок, коли $n \geq 0$ та проведемо доведення за МІ (\textit{детально МІ розписувати не буду, тут цілком все прозоро}).
\begin{align*}
[x,y]^{n+1} & = [x,y]^n [x,y] = [x,y]^n xyx^{-1}y^{-1} \overset{\text{$x$ комутує з $[x,y]$}}{=} x [x,y]^n yx^{-1}y^{-1} \\ & \overset{\text{припущення МІ}}{=} x[x^n,y]yx^{-1}y^{-1} = x(x^nyx^{-n}y^{-1})yx^{-1}y^{-1} = [x^{n+1},y].
\end{align*}
Аналогічно, користуючись умовою, що $y$ (\textit{тому й $y^{-1}$}) комутує з $[x,y]$, одержимо
\begin{align*}
[x,y]^{n+1} = [x,y][x,y]^n = xyx^{-1}y^{-1} [x,y]^n = \dots = [x,y^{n+1}].
\end{align*}
Тепер розглянемо випадок, коли $n < 0$. Зауважимо, що
\begin{align*}
x [x,y] = [x,y] x \implies xyx^{-1}y^{-1} = yx^{-1}y^{-1}x \implies [x,y]^{-1} = [y,x^{-1}]^{-1} = [x^{-1}y].
\end{align*}
Аналогічно із $y[x,y] = [x,y]y$ випливатиме, що $[x,y]^{-1} = [x,y^{-1}]$.
\end{enumerate}

\item \textit{Доведення другої формули}.\\
Знову доведення буде за МІ. Одразу напишемо
\begin{align*}
(xy)^{n+1} & = (xy)^n (xy) \overset{\text{припущення МІ}}{=} [y,x]^{C_n^2} x^n y^n xy = [y,x]^{C_n^2} x^{n+1} [x^{-1},y^n] y^{n+1} \\
& \overset{\text{формули І}}{=} [y,x]^{C_n^2} x^{n+1} [y,x]^n y^{n+1} \overset{\text{$x$ комутує з $[y,x]$}}{=} [y,x]^{C_n^2} [y,x]^n x^{n+1} y^{n+1} = [y,x]^{C_{n+1}^2} x^{n+1} y^{n+1}.
\end{align*}
У двох випадках довели все за МІ.
\end{proof}

\begin{lemma}
Нехай $G$ --- скінченна група та $n > 1$, причому $(xn)^n = x^n y^n$ для всіх $x,y \in G$. Позначимо
\[
G[n] = \{g \in G: g^n = 1\}, \qquad G^n = \{ g^n \mid g \in G\}.
\]
Тоді $G[n], G^n \triangleleft G$. Крім того, $|G^n| = [G:G[n]]$.\\
(\textit{була вправа в базовій теорії груп})
\end{lemma}

\subsection{Про підгрупи $p$-груп}
\begin{theorem}
Нехай $G$ --- (\textit{скінченна}) $p$-група, що містить єдину підгрупу порядка $p$ та більш ніж одну циклічну підгрупу індекса $p$. Тоді $G \cong Q_8$.
\end{theorem}

\begin{proof}
Нехай $A$ --- підгрупа $G$ індекса $p$. Зауважимо, що $A \triangleleft G$. Звідси $G/_A$ --- група порядка $p$, але звідси $x^p \in A$ для будь-якого $x \in G$.\\
Нехай $A = \left<a\right>,\ B = \left<b\right>$ будуть різними підгрупами індекса $p$. Позначимо $D = A \cap B$ та зауважимо, що $D \triangleleft G$. Із міркувань вище отримали, що $G^p = \{x^p \mid x \in G \} \supset D$.\\
Оскільки $A,B$ --- різні максимальні підгрупи, то $G = AB$. Із формули добутків випливає, що $[G:D] = p^2$. Звідси $G/_D$ --- абелева, тому за основною теоремою про комутанти $G' \leq D$. Крім того, $D \leq Z(G)$ (\textit{це неважко показати}).\\
Отже, $G' \leq D \leq Z(G)$, тобто будь-який комутант комутує з елементами групи $G$. Тоді $[y,x]^p = [y^p,x]$ для всіх $x,y \in G$. Проте оскільки $y^p \in D \leq Z(G)$, то тоді $[y,x]^p = e$. Також відомо, що $(xy)^p = [y,x]^{C_p^2} x^py^p$.\\
!Припустимо, що $p$ --- непарне число. Тоді $p \mid C_p^2$, звідси одержимо $(xy)^p = x^py^p$. Проте остання лема каже, що
\[
|G[p]| = [G:G^p] = [G:D] [D:G^p] \geq p^2.
\]
Тоді $G[p]$ містить підгрупу порядка $p^2$. Два варіанти: або це $\mathbb{Z}_{p^2}$ (\textit{просто абелева}), або $\mathbb{Z}_p \times \mathbb{Z}_p$ (\textit{елементарна абелева}). Оскільки елементи $G[p]$ мають порядок не більше за $p$, то нам підходить лише елементарно абелева підгрупа. Отже, $G[p]$ (\textit{внаслідок чого й $G$}) містить більше, ніж одну підгрупу порядка $p$ --- суперечність!
\bigskip \\
Отже, $p = 2$, а також $[G:D] = 4$. Тоді $D$ --- група порядка $2$, а тому
\[
D = \left<a^2\right> \overset{\substack{G^2 \subset D \\ a^2 \in G^2}}{=} G^2 \leq Z(G).
\]
Внаслідок чого $[y,x]^2 = [y,x^2] = e$. Звідси вивпливає, що $(xy)^4 = [y,x]^6 x^4 y^4 = x^4 y^4$, а отже
\[
|G[4]| = [G:G^4] = [G:D] [D:G^4] = 8,
\]
просто тому що неважко зауважити, що $G^4 = \left<a^4\right>$. Нарешті, за умовою $G$ містить лише одну підгрупу порядка $2$, тобто для $G$ (\textit{зокрема й для $G[4]$}) існує лише одна інволюція. Значить, $G[4]$ зобов'язана містити дві різні циклічні підгрупи $\left<u\right>, \left<v\right>$ порядка $4$ (\textit{якби існувала лише єдина циклічна підгрупа порядка $4$, то $G[4]$ мала б більше за одну інволюцію}).\\
!Припустимо, що $a^4 \neq e$. Можемо тоді обрати підгрупу $\left<u\right> \leq D = \left<a^2\right> \leq Z(G)$ (TODO: чому саме так?), але тоді $\left<u\right>\left<v\right>$ стане абелевою підгрупою $G$. Саме ця підгрупа містить рівно дві інволюції: або $u^2$ та $v^2$ (\textit{у випадку $u^2 \neq v^2$}), або $u^2$ та $uv^{-1}$ (\textit{у випадку $u^2 = v^2$, причому точно $uv^{-1} \neq u^2$}). Оскільки $G[4]$ має лише одну інволюцію, то ми прийшли до суперечності!\\
Коротше, $a^4 = e$, тобто $|D| = 2$ та $|G| = 8$. Оскільки група $G$ має лише одну інволюцію, то або $G \cong Q_8$, або $G \cong \mathbb{Z}_8$. Саме $Q_8$ має більше, ніж одну підгрупу індекса $2$.
\end{proof}

\begin{theorem}
Нехай $G$ --- скінченна $p$-група, що містить єдину підгрупу порядка $p$. Тоді $G$ --- циклічна або є узагальненим кватерніоном.
\end{theorem}

\begin{proof}
Доведення проведемо за МІ по $n$, де $|G| = p^n$.
\begin{enumerate}[wide=0pt,label={\Roman*.}]
\item \textit{$p$ --- непарне просте}.\\
При $n > 0$ отримаємо, що $G$ містить підгрупу $H$ індекса $p$. За припущенням МІ $H$ буде циклічною (\textit{не узагальненим кватерніоном, бо там порядок $2^n$}). Не існує інших підгруп індекса $p$ (\textit{інакше за попередньою теоремою $G \cong Q_8$, що неможливо}). Отже, $H$ --- єдина максимальна підгрупа $G$, тобто $H$ містить будь-яку іншу власну підгрупу $G$. У цьому випадку $G$ обов'язково стане циклічною. Адже в інакшому випадку $\left<x\right>$ буде власною підгрупою для всіх $x \in G$, а тому $G \leq H$, що неможливо.

\item \textit{$p = 2$}.\\
Можемо одразу припускати, що $G$ неабелева (\textit{бо тоді $G$ стане циклічною --- і закінчили}). Нехай $A \triangleleft G$ --- максимальна абелева. Оскільки $A$ містить єдину інволюцію (\textit{як і група $G$}), то $A$ циклічна, скажімо $A = \left<a\right>$. Хочемо довести, що $[G:A] = 2$.\\
!Припустимо, що $[G:A] \neq 2$, тобто $|G/_A \geq 4$. Для суперечності треба багато роботи зробити.
\begin{enumerate}[wide=0pt,label={\roman*.}]
\item \textit{Існує елемент в квадраті в групі $G/_A$}.\\
Якщо не існує елемента в квадраті в $G/_A$, то тоді існує $Ab \in G/_A$ так, що $b^2 \notin A$. Розглянемо тепер підгрупу $H = \left<a,b^2\right> < \left<a,b\right> \leq G$. Якщо раптом $H$ --- абелева, то тоді $b^2 \in C_G(A) = A$, що неможливо (TODO: вставити Th 5.41). Тоді за припущенням МІ група $H$ має бути узагальненим кватерніоном. Можемо тоді припустити, що $b^2 a b^{-2} = a^{-1}$ (TODO: чому не навпаки?). Оскільки відомо, що $\left<a\right> \triangleleft G$, то це дає $bab^{-1} = a^{i}$ для деякого $i$, а тому
\[
a^{-1} = b^2 a b^{-2} = b(bab^{-1})b^{-1} = ba^{i}b^{-1} = a^{i^2},
\]
внаслідок чого $i^2 \equiv -1 \pmod {2^e}$, де число $\ord a = 2^e$. Оскільки $A > Z(G)$, то тоді $e \geq 2$.\\
Якщо $e = 2$, то тоді $i^2 \equiv -1 \pmod 4$, але ж число $-1$ не є квадратичним лишком за $\!\!\pmod 4$.\\
Якщо $e \geq 3$, то $i^2 \equiv -1 \pmod {2^e}$ не виконується (TODO: подумати).\\
Отже, ми прийшли до суперечності припущенню, що не існує елемента в квадраті в $G/_A$.

\item \textit{(дописати)}.\\
Отже, $G/_A$ має елемент елемент в квадратів. Оскільки $|G/_A| = 4$, то $G/_A$ має містити підгрупу Кляйна $V$ (TODO:?). Тому існують елементи $c,d$ так, що $c,d,c^{-1}d \notin A$, причому $\left<a,c\right>, \left<a,d\right>, \left<a,c^{-1}d\right>$ будуть власними підгрупами $G$. Жодна з них не буде абелевою, інакше або $c$, або $d$, або $c^{-1}d$ потраплять в $C_G(A) = A$. Нарешті, $c,d,c^{-1}d$ є узагальненими кватерніонами (TODO:?). Проте $cac^{-1} = a^{-1} = dad^{-1}$ дає $c^{-1}d \in C_G(A)$ --- нарешті суперечність!
\end{enumerate}
Все, віднині $[G:A] = 2$. Значить, якщо $b \in G$, то завжди $b^2 \in A = \left<a\right>$. Можемо припустити, що існує число $r \leq n-2$ таке, що $b^2 = a^{2^r}$ (TODO: вопще не зрозумів). Оскільки $\left<a\right> \triangleleft G$, то $bab^{-1} = a^t$. Оскільки $G$ має лише одну інволюцію, то $t = \pm 1$ (\textit{TODO: вставити доведення 5.45}). Однак $t \neq 1$, інакше $a,b$ комутуватимуть, а тому $G$ стане абелевою, тож циклічною. Можна тоді припускати, що $t = -1$ та $G = \left<a,b\right>$, де
\[
a^{2^{n-1}} = 1, \qquad bab^{-1} = a^{-1}, \qquad b^2 = a^{2^r}.
\]
Залишилося довести, що $r = n-2$. Оскільки $t = -1$, то $2^r \equiv -2^{r} \pmod {2^{n-1}}$, звідси $2^{2+1} \equiv 0 \pmod {2^{n-1}}$, тобто $r = n-2$. \qedhere
\end{enumerate}
\end{proof}

\section*{Задачі, які я ще не розв'язав}
\begin{itemize}[wide=0pt]
\item $S_n$ при $n \geq 5$ не має підгрупи $H$ з індексом $2 < [S_n:H] < n$ (\textit{розв'язана}).
\iffalse
\bigskip \\
Припустимо, що $H \leq S_n$ із індексом $t = [S_n:H]$, де $2 < t < n$. Оберемо гомоморфізм $\rho \colon S_n \to S_t$, причому $\ker \varphi \leq H$. Оскільки $\ker \varphi \triangleleft S_n$, то тоді або $\ker \varphi = \{\varepsilon\}$, або $\ker \varphi = A_n$ --- інших нема.\\
Якщо $\ker \varphi = \{\varepsilon\}$, то гомоморфізм $\rho \colon S_n \to S_t$ --- ін'єктивний --- неможливо.\\
Якщо $\ker \varphi = A_n$, то тоді $[S_n:A_n] = [S_n:H] \cdot [H:A_n]$, тобто $2 = t \cdot [H:A_n]$ --- неможливо.
\fi

\item (\textit{Теорема Бернсайда}). Нехай $p,q$ --- прості числа. Тоді будь-яка група порядка $p^m q^n$ розв'язна. Для початку треба довести, що якщо $G$ --- скінченна та проста, то $|a^G| \neq p^k > 1$ для жодного класу (\textit{але останнє, здається, доводиться через теорію зображень}).
\iffalse
\bigskip \\
Тепер нехай $|G| = p^m q^n$. Припустимо, що $G$ --- проста, тоді $G$ --- неабелева (\textit{інакше ми би обрали підгрупу Сілова, яка була б нормальною}). Отже, $Z(G) = e$. Маємо формулу
\[
|G| = |Z(G)| + \sum_{a} |a^G|,
\]
проте оскільки $G$ --- проста, то $pq \mid |a^G|$, а тому звідси $0 \equiv 1 + 0 \pmod {pq}$, що неможливо.\\
Отже, $G$ --- непроста, тому підгрупа $\{e\} \triangleleft G' \triangleleft G$. Можна далі сказати за МІ, що $G'$ буде розв'язною.
\fi

\item Нехай $\gcd([G:H],[G:K]) = 1$. Тоді $[G: H\cap K] = [G:H][G:K]$ (\textit{розв'язана, лише коли $G$ --- скінченна}).
\iffalse
\bigskip \\
Оскільки $\gcd([G:H],[G:K]) = 1$, то тоді $\lcm([G:H],[G:K]) = [G:H] [G:K]$. Іншими словами, хочемо довести, що $[G: H \cap K]$ буде спільним кратним $[G:H] [G:K]$, причому найменшим.\\
Дійсно, зауважимо, що виконується така рівність:
\[
[G: H \cap K] = [G: H] [H : H \cap K], \qquad [G: H \cap K] = [G: K] [K : H \cap K].
\]
Звідси випливає, що $[G:H],\ [G:K] \mid [G: H \cap K]$, тому щонайменше $[G:H \cap K] \geq [G:H] [G:K]$.\\
Із іншого боку, справедливе наступне:
\[
[G:H \cap K] = \dfrac{|G|}{|H \cap K|} = \dfrac{|G| |HK|}{|H| |K|} \leq \dfrac{|G| |G|}{|H| |K|} = [G:H] [G:K].
\]
Значить, дійсно отримаємо $[G: H\cap K] = [G:H][G:K]$.
\fi

\item $Q_8,D_4$ --- єдині неабелеві групи порядка $8$. (\textit{десь залишити доведення})

\item Якщо $G$ --- група порядка $8$ з однією інволюцією, то ця група або $Q_8$, або $\mathbb{Z}_8$ (\textit{тут все легко})
\end{itemize}
